# The Debian and Ubuntu packages a Linux KVM host that builds and runs cladefs's tests needs: what
# scripts/host-packages.sh installs, which a hotbed base's setup step and host_setup
# (scripts/lib/linux-host-setup.sh) run. A word per package; `#` starts a comment.

build-essential pkg-config python3 curl ca-certificates git rsync e2fsprogs sqlite3
# clang's libraries build Firecracker's bindings; guest/sim and aarch64 Linux Rust builds
# (.cargo/config.toml) link with lld (-fuse-ld=lld); guest/sim builds with ASan and UBSan, whose
# headers and runtimes libclang-rt-dev holds.
clang libclang-dev llvm lld libclang-rt-dev
# The guest's static binaries.
musl-tools
# Firecracker's seccompiler.
libseccomp-dev
# The root filesystems of guests that need one, and the image of fxfs-guest-docker-dax's
# containers.
podman busybox-static
# The guest kernel (scripts/build-guest-kernel.sh).
flex bison bc cpio libelf-dev libssl-dev dwarves
# eu-stack, which saves a hung test's stacks (the guest tests' runner, TASK-44.13).
elfutils
# spatch, which checks the guest's C (scripts/cocci-guest.py, TASK-44.17).
coccinelle
# fusermount3, which mounts FUSE filesystems for users other than root, as cladefsd's mount of
# cells does (TASK-222, TASK-475.2).
fuse3
